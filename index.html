// 4 AÇIK KARTLI SESLENDİRME MOTORU (TABLET & MOBİL İÇİN AKUSTİK GARANTİLİ) function pickAndPlayVoice(voiceType) { if (!('speechSynthesis' in window)) return; if (currentlyPlayingVoiceType === voiceType) { stopReadingAloud(); return; } stopReadingAloud(); const currentText = textChunks[currentChunkIndex] || document.getElementById('textInput').value.trim(); if (!currentText) return; const utterance = new SpeechSynthesisUtterance(currentText); const all = window.speechSynthesis.getVoices(); let matchVoice = null; // 1. Kadın (Amerikan) - Canlı, yumuşak peri tonu if (voiceType === 'US_F') { utterance.lang = 'en-US'; utterance.pitch = 1.2; // Belirgin kadın/peri tonu utterance.rate = 0.92; matchVoice = all.find(v => (v.lang === 'en-US' || v.lang === 'en_US') && (v.name.includes('Female') || v.name.includes('Samantha') || v.name.includes('Zira') || v.name.includes('Karen') || v.name.includes('Victoria'))); } // 2. Kadın (İngiliz) - Sakin, zarif ton else if (voiceType === 'UK_F') { utterance.lang = 'en-GB'; utterance.pitch = 1.05; // Klasik sakin kadın tonu utterance.rate = 0.88; matchVoice = all.find(v => (v.lang === 'en-GB' || v.lang === 'en_GB') && (v.name.includes('Female') || v.name.includes('Hazel') || v.name.includes('Susan') || v.name.includes('Serena') || v.name.includes('Fiona'))); } // 3. Erkek (Amerikan) - Derin, enerjik erkek tonu else if (voiceType === 'US_M') { utterance.lang = 'en-US'; utterance.pitch = 0.70; // TABLETTE BİLE SESİ NET ERKEK YAPAR utterance.rate = 0.90; matchVoice = all.find(v => (v.lang === 'en-US' || v.lang === 'en_US') && (v.name.includes('Male') || v.name.includes('David') || v.name.includes('Mark') || v.name.includes('Alex') || v.name.includes('Aaron'))); } // 4. Erkek (İngiliz) - Bilge, derin büyücü erkek tonu else if (voiceType === 'UK_M') { utterance.lang = 'en-GB'; utterance.pitch = 0.62; // TABLETTE BİLE DERİN TOK ERKEK SESİ YAPAR utterance.rate = 0.85; matchVoice = all.find(v => (v.lang === 'en-GB' || v.lang === 'en_GB') && (v.name.includes('Male') || v.name.includes('George') || v.name.includes('Oliver') || v.name.includes('Daniel') || v.name.includes('Arthur'))); } // Eğer tablette o özel ses ismi yoksa, en azından dil ailesini eşle (pitch tonu erkek/kadın ayrımını çözer) if (!matchVoice) { if (voiceType.startsWith('UK')) { matchVoice = all.find(v => v.lang.includes('GB') || v.lang.includes('UK')); } if (!matchVoice) { matchVoice = all.find(v => v.lang.startsWith('en')); } } if (matchVoice) utterance.voice = matchVoice; utterance.onend = () => { stopReadingAloud(); }; window.speechSynthesis.speak(utterance); currentlyPlayingVoiceType = voiceType; document.querySelectorAll('.voice-pick-card').forEach(c => c.classList.remove('active')); const activeCard = document.getElementById(`card_${voiceType}`); if (activeCard) activeCard.classList.add('active'); const activeIcon = document.getElementById(`icon_${voiceType}`); if (activeIcon) activeIcon.innerText = "⏹"; }
